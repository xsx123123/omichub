> 优化项：

## 一、用户昵称首字符头像

### 效果说明
将聊天界面中固定的「我」文字标识，替换为当前登录用户的昵称首字符，并做圆形头像化展示。

例如：用户昵称为 **JIAN** → 显示 **J**；用户昵称为 **张三** → 显示 **张**。

### React 组件代码

```tsx
// components/UserAvatar.tsx
import React from 'react';
import { Avatar } from '@arco-design/web-react';

interface UserAvatarProps {
  nickname?: string;      // 用户昵称，从 userStore 或 auth context 获取
  size?: number;          // 头像尺寸，默认 32
  className?: string;
}

const UserAvatar: React.FC<UserAvatarProps> = ({
  nickname = '用户',
  size = 32,
  className,
}) => {
  // 获取首字符（支持中英文）
  const getInitial = (name: string): string => {
    if (!name || name.trim().length === 0) return '?';
    const trimmed = name.trim();
    // 取第一个字符
    const firstChar = trimmed.charAt(0);
    // 如果是英文字母，转大写
    if (/[a-zA-Z]/.test(firstChar)) {
      return firstChar.toUpperCase();
    }
    // 中文或其他字符直接返回
    return firstChar;
  };

  // 根据昵称生成稳定的背景色（避免所有用户都是灰色）
  const generateColor = (name: string): string => {
    const colors = [
      '#165DFF', '#14C9C2', '#F7BA1E', '#F53F3F',
      '#722ED1', '#EB2F96', '#00B42A', '#3491FA',
    ];
    let hash = 0;
    for (let i = 0; i < name.length; i++) {
      hash = name.charCodeAt(i) + ((hash << 5) - hash);
    }
    const index = Math.abs(hash) % colors.length;
    return colors[index];
  };

  const initial = getInitial(nickname);
  const bgColor = generateColor(nickname);

  return (
    <Avatar
      size={size}
      style={{
        backgroundColor: bgColor,
        color: '#fff',
        fontWeight: 600,
        fontSize: size * 0.5,
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        userSelect: 'none',
      }}
      className={className}
    >
      {initial}
    </Avatar>
  );
};

export default UserAvatar;
```

## 三、进阶：更丰富的 Canvas 粒子动画

如果纯 CSS 粒子不够看，可以用轻量级 Canvas 实现更炫的全息效果：

```tsx
// components/HologramParticles.tsx
import React, { useEffect, useRef } from 'react';

const HologramParticles: React.FC = () => {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const resize = () => {
      canvas.width = canvas.offsetWidth;
      canvas.height = canvas.offsetHeight;
    };
    resize();
    window.addEventListener('resize', resize);

    // 粒子配置
    const particles: Array<{
      x: number; y: number; vx: number; vy: number;
      size: number; alpha: number; text: string;
    }> = [];

    const texts = ['0', '1', 'A', 'T', 'G', 'C', '✦'];

    for (let i = 0; i < 30; i++) {
      particles.push({
        x: Math.random() * canvas.width * 0.5 + canvas.width * 0.4,
        y: Math.random() * canvas.height * 0.6 + canvas.height * 0.2,
        vx: (Math.random() - 0.5) * 0.3,
        vy: -Math.random() * 0.5 - 0.2,
        size: Math.random() * 12 + 8,
        alpha: Math.random() * 0.3 + 0.1,
        text: texts[Math.floor(Math.random() * texts.length)],
      });
    }

    let animId: number;
    const animate = () => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      particles.forEach((p) => {
        p.x += p.vx;
        p.y += p.vy;
        p.alpha -= 0.003;

        // 重置飘出屏幕的粒子
        if (p.alpha <= 0 || p.y < -20) {
          p.x = Math.random() * canvas.width * 0.5 + canvas.width * 0.4;
          p.y = canvas.height * 0.8 + Math.random() * 50;
          p.alpha = Math.random() * 0.3 + 0.1;
          p.vy = -Math.random() * 0.5 - 0.2;
        }

        ctx.font = `${p.size}px "Courier New", monospace`;
        ctx.fillStyle = `rgba(100, 200, 255, ${p.alpha})`;
        ctx.fillText(p.text, p.x, p.y);
      });

      animId = requestAnimationFrame(animate);
    };
    animate();

    return () => {
      window.removeEventListener('resize', resize);
      cancelAnimationFrame(animId);
    };
  }, []);

  return (
    <canvas
      ref={canvasRef}
      style={{
        position: 'absolute',
        right: 0,
        bottom: 0,
        width: '55%',
        height: '70%',
        pointerEvents: 'none',
        zIndex: 0,
      }}
    />
  );
};

export default HologramParticles;
```

在组件中使用：
```tsx

  <HologramParticles />  {/* 替换纯 CSS 粒子层 */}
  <div className="shaonu-chat-messages">...</div>
</div>
```
